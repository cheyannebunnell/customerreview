/** * Survey backend (Google Apps Script) * * SETUP * 1. Project Settings > Script Properties > add GEMINI_API_KEY. * 2. Make sure the spreadsheet has a tab named "Responses" (headers are * created automatically if the tab is empty). * 3. Deploy > New deployment > Web app * Execute as: Me | Who has access: Anyone * After ANY code change, create a NEW deployment version, otherwise the * /exec URL keeps serving the old code. */ const SHEET_NAME = "Responses"; // Change this if Google retires the model. See https://ai.google.dev/gemini-api/docs/models const GEMINI_MODEL = "gemini-2.5-flash"; const HEADERS = [ "Timestamp", "Usage Frequency", "Favorite Feature", "Satisfaction", "Free Response", "AI Analysis" ]; const PENDING = "Pending"; const MAX_TEXT_LENGTH = 4000; /** * GET Requests */ function doGet(e) { try { const action = e && e.parameter ? e.parameter.action : ""; if (action === "results") { return getResults(); } return jsonResponse({ status: "ok", message: "Survey API Running" }); } catch (error) { return jsonResponse({ status: "error", message: error.message }); } } /** * POST Requests */ function doPost(e) { const lock = LockService.getScriptLock(); try { if (!e || !e.postData || !e.postData.contents) { throw new Error("Empty request body"); } const data = JSON.parse(e.postData.contents); // Basic validation if (!data.usageFrequency || !data.favoriteFeature || !data.satisfaction || !data.freeResponse) { throw new Error("Missing required fields"); } const freeResponse = String(data.freeResponse).slice(0, MAX_TEXT_LENGTH); // Serialize writes so simultaneous submissions don't interleave lock.waitLock(30000); const sheet = getSheet(); // Save the row immediately with a placeholder. The (slow) Gemini call // happens afterwards in a background trigger, so the user isn't kept waiting. sheet.appendRow([ new Date(), String(data.usageFrequency), String(data.favoriteFeature), String(data.satisfaction), freeResponse, PENDING ]); lock.releaseLock(); // The row is already saved, so a trigger failure (e.g. permission not yet // approved) must not make the submission look like it failed. try { scheduleAnalysis(); } catch (err) { console.error("Could not schedule analysis: " + err.message); } return jsonResponse({ status: "success" }); } catch (error) { try { lock.releaseLock(); } catch (ignore) {} console.error(error); return jsonResponse({ status: "error", message: error.message }); } } /** * Get Chart Data */ function getResults() { const sheet = getSheet(); const lastRow = sheet.getLastRow(); const usageCounts = {}; const featureCounts = {}; const satisfactionCounts = {}; // Row 1 is always the header (guaranteed by getSheet), so data starts at row 2 if (lastRow >= 2) { const rows = sheet.getRange(2, 1, lastRow - 1, 4).getValues(); rows.forEach(row => { const usage = row[1]; const feature = row[2]; const satisfaction = row[3]; if (usage) usageCounts[usage] = (usageCounts[usage] || 0) + 1; if (feature) featureCounts[feature] = (featureCounts[feature] || 0) + 1; if (satisfaction) satisfactionCounts[satisfaction] = (satisfactionCounts[satisfaction] || 0) + 1; }); } return jsonResponse({ usageFrequency: usageCounts, favoriteFeature: featureCounts, satisfaction: satisfactionCounts }); } /** * Background analysis */ // Creates a one-off trigger (only if one isn't already waiting) function scheduleAnalysis() { const exists = ScriptApp.getProjectTriggers() .some(t => t.getHandlerFunction() === "processPending"); if (!exists) { ScriptApp.newTrigger("processPending") .timeBased() .after(1000) .create(); } } // Analyzes every row still marked "Pending" function processPending() { // Remove the one-off trigger(s) that launched this run ScriptApp.getProjectTriggers().forEach(t => { if (t.getHandlerFunction() === "processPending") { ScriptApp.deleteTrigger(t); } }); const sheet = getSheet(); const lastRow = sheet.getLastRow(); if (lastRow < 2) return; const range = sheet.getRange(2, 5, lastRow - 1, 2); // columns E:F const values = range.getValues(); values.forEach((row, i) => { if (row[1] !== PENDING) return; let analysis; try { analysis = analyzeWithGemini(row[0]); } catch (error) { analysis = "Gemini Error: " + error.message; } // Write only this one cell so we never clobber other rows sheet.getRange(i + 2, 6).setValue(analysis); }); } /** * Gemini Analysis */ function analyzeWithGemini(text) { const apiKey = PropertiesService .getScriptProperties() .getProperty("GEMINI_API_KEY"); if (!apiKey) { throw new Error("GEMINI_API_KEY script property is not set"); } const url = "https://generativelanguage.googleapis.com/v1beta/models/" + GEMINI_MODEL + ":generateContent"; const payload = { contents: [ { parts: [ { text: "Analyze this customer feedback and provide:\n" + "1. Sentiment\n" + "2. Key Themes\n" + "3. Recommendations\n\n" + "Feedback:\n" + text } ] } ] }; const response = UrlFetchApp.fetch(url, { method: "post", contentType: "application/json", // Key goes in a header so it doesn't end up in URLs/logs headers: { "x-goog-api-key": apiKey }, payload: JSON.stringify(payload), muteHttpExceptions: true }); const responseCode = response.getResponseCode(); const responseText = response.getContentText(); if (responseCode !== 200) { throw new Error("HTTP " + responseCode + ": " + responseText.slice(0, 300)); } const result = JSON.parse(responseText); // Guard against blocked/empty responses const parts = result.candidates && result.candidates[0] && result.candidates[0].content && result.candidates[0].content.parts; if (!parts || !parts.length) { throw new Error("Gemini returned no content"); } return parts.map(p => p.text || "").join(""); } /** * Helpers */ // Returns the Responses sheet, creating the header row if needed function getSheet() { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName(SHEET_NAME); if (!sheet) { throw new Error('Sheet "' + SHEET_NAME + '" not found'); } if (sheet.getLastRow() === 0) { sheet.appendRow(HEADERS); } return sheet; } function jsonResponse(obj) { return ContentService .createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); }
